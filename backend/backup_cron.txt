# Para agendar backup diário às 12h (meio-dia) horário local, adicione esta linha ao seu crontab
# substituindo o caminho pelo diretório do repositório na sua máquina.
# Exemplo para este repositório em /home/bruno/panificadora:
0 12 * * * cd /home/bruno/panificadora && bash backend/backup_db.sh >> backend/backups/backup_cron.log 2>&1
